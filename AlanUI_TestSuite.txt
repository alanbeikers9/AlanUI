    local v = valores[math.random(1, #valores)]
    lblSimple:Set(v)

end })

    local nuevo = (actual + 10) % 110
    prog:Set(nuevo > 100 and 0 or nuevo)
end })

            a = paso("duplicar repetido falla", gestor:Duplicar("zz_t_origen", "zz_t_copia") == false) and a
            a = paso("renombrar", gestor:Renombrar("zz_t_copia", "zz_t_renom") == true
                and isfile(ruta("zz_t_renom")) and not isfile(ruta("zz_t_copia"))) and a
            local texto = gestor:Exportar("zz_t_origen")
            local decodifica = type(texto) == "string" and (pcall(function() return game:GetService("HttpService"):JSONDecode(texto) end))
            a = paso("exportar", decodifica) and a
            a = paso("importar", gestor:Importar("zz_t_import", texto) == true and isfile(ruta("zz_t_import"))) and a
            a = paso("importar sin sobreescribir falla", gestor:Importar("zz_t_import", texto) == false) and a
            a = paso("importar sobreescribiendo", gestor:Importar("zz_t_import", texto, true) == true) and a
            a = paso("importar texto invalido falla", gestor:Importar("zz_t_import2", "esto no es json") == false
                and not isfile(ruta("zz_t_import2"))) and a
            a = paso("nombre invalido falla", gestor:Duplicar("zz_t_origen", "a/b") == false) and a
            limpiar()
            return a, table.concat(pasos, " | ")
        end)

        prueba("Profiler se ejecuta sin error", function()
            AlanUI:Profiler()
            return true
        end)

        local fallos = 0
        for _, r in ipairs(resultados) do
            if r:find("FAIL", 1, true) then fallos = fallos + 1 end
        end
        print(string.format("[TEST] Pruebas terminadas: %d total, %d fallos", #resultados, fallos))
        AlanUI:Notificar({
            Titulo = fallos == 0 and "✅ Pruebas OK" or ("🔴 " .. fallos .. " fallos"),
            Texto = "Pulsa 📥 Guardar log para exportar el resultado",
            Duracion = 5,
        })
        ejecutando = false
    end)
end
